cmake_minimum_required(VERSION 3.22)
project(BqstGritEquivalence LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_subdirectory("${JUCE_ROOT}" juce EXCLUDE_FROM_ALL)
add_library(runtime STATIC)
target_link_libraries(runtime PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_lto_flags)
target_include_directories(runtime PUBLIC "${JUCE_ROOT}/modules")
target_compile_definitions(runtime PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_STANDALONE_APPLICATION=1)
foreach(version pre current)
    foreach(lab 0 1)
        set(target ${version}_lab${lab})
        add_executable(${target} "${HARNESS}" "${SNAPSHOTS}/${version}/src/PluginProcessor.cpp" "${SNAPSHOTS}/${version}/src/BqtParameterLayout.cpp" "${SNAPSHOTS}/${version}/src/BqtProcessorDsp.cpp")
        target_include_directories(${target} PRIVATE "${SNAPSHOTS}/${version}/src")
        target_compile_definitions(${target} PRIVATE BQST_HEADLESS_TESTS=1 BQST_GRIT_LAB=${lab} JucePlugin_Name="BQST")
        target_link_libraries(${target} PRIVATE runtime)
    endforeach()
endforeach()
